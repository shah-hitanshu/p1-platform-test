Design tokens

Install with pnpm dlx shadcn@latest add @p1/tokens. The file lands in app/p1-tokens.css and is yours to modify.

p1-tokens.css

/* P1 marketing design tokens.
 *
 * Values derive from PDS v2 primitives where the concept exists there, with a
 * literal fallback in every case: a customer project is not guaranteed to load
 * the PDS token layer, and an unqualified var(--pds-…) renders as nothing.
 *
 * This file is installed into your project by `shadcn add @p1/tokens` and it is
 * yours. Override anything:
 *
 *   :root { --p1-accent: #7c3aed; --p1-radius-card: 0; }
 *
 * Breakpoints are NOT tokens — the ladder is 40rem / 48rem / 64rem / 80rem,
 * applied literally in media queries.
 */

:root {
  /* Surfaces and ink */
  --p1-surface-default: var(--pds-color-surface-default, oklch(1 0 0));
  --p1-surface-subtle: var(--pds-color-surface-default-secondary, oklch(0.9851 0 0));
  --p1-surface-inverse: var(--pds-color-surface-reverse, oklch(0.2046 0 0));
  --p1-fg-default: var(--pds-color-foreground-default, oklch(0.2046 0 0));
  --p1-fg-muted: var(--pds-color-foreground-default-secondary, oklch(0.5103 0 0));
  --p1-fg-inverse: var(--pds-color-foreground-reverse, oklch(1 0 0));
  --p1-border: var(--pds-color-border-default, oklch(0.9401 0 0));
  --p1-border-strong: var(--pds-color-border-default-secondary, oklch(0.8853 0 0));

  /* Dark surface — near-black with slight blue cast (the "dark" tone) */
  --p1-surface-dark: oklch(0.17 0.01 270);


  /* Accent — the tone every "branded" block variant resolves to. Replaces the
     hard-coded indigo/purple the blocks used, which is why `tone: "indigo"`
     becomes `tone: "accent"` in this phase. */
  --p1-accent: var(--pds-color-interactive-link-default, oklch(0.55 0.21 280));
  --p1-accent-hover: var(--pds-color-interactive-link-hover, oklch(0.48 0.20 280));
  --p1-on-accent: var(--pds-color-foreground-reverse, oklch(1 0 0));

  /* Status */
  --p1-success: var(--pds-color-status-success-foreground, oklch(0.62 0.15 152));
  --p1-warning: var(--pds-color-status-warning-foreground, oklch(0.75 0.15 80));
  --p1-critical: var(--pds-color-status-critical-foreground, oklch(0.58 0.22 27));

  /* Gradient stops — for the gradient/banner tones */
  --p1-gradient-from: oklch(0.65 0.27 329);
  --p1-gradient-via: oklch(0.55 0.24 292);
  --p1-gradient-to: oklch(0.34 0.18 265);

  /* On-dark foreground hierarchy */
  --p1-fg-inverse-muted: oklch(1 0 0 / 0.65);
  --p1-fg-inverse-subtle: oklch(1 0 0 / 0.8);
  --p1-fg-inverse-faint: oklch(1 0 0 / 0.9);

  /* On-dark borders */
  --p1-border-inverse: oklch(1 0 0 / 0.15);
  --p1-border-inverse-strong: oklch(1 0 0 / 0.3);

  /* Subtle separator — uses PDS border token */
  --p1-border-subtle: var(--pds-color-border-default, oklch(0.925 0 0));

  /* On-dark input chrome */
  --p1-input-inverse-bg: oklch(1 0 0 / 0.08);
  --p1-input-inverse-bg-hover: oklch(1 0 0 / 0.15);

  /* Glow decoration */
  --p1-glow-warm: oklch(1 0 0 / 0.5);
  --p1-glow-soft: oklch(1 0 0 / 0.17);

  /* Overlays */
  --p1-overlay: oklch(0 0 0 / 0.6);
  --p1-overlay-subtle: oklch(0 0 0 / 0.1);

  /* Code inline surface */
  --p1-surface-code: oklch(0 0 0 / 0.05);


  /* Spacing */
  --p1-space-xs: 0.25rem;
  --p1-space-sm: 0.5rem;
  --p1-space-md: 1rem;
  --p1-space-lg: 1.5rem;
  --p1-space-xl: 2rem;
  --p1-space-2xl: 3rem;
  --p1-space-3xl: 4rem;

  /* Section rhythm and responsive page gutters — the px-6 / sm:px-10 / lg:px-16
     ladder the blocks used. Blocks reassign the gutters in media queries. */
  --p1-section-y: 4rem;
  --p1-gutter-sm: 1.5rem;
  --p1-gutter-md: 2.5rem;
  --p1-gutter-lg: 4rem;

  /* Shared block padding. MUST equal apps/p1-starter's blockPaddingClass
     ("px-16 py-6"), or installed pages misalign where registry blocks sit next
     to starter-kit blocks. Asserted by tokens/tokens.test.ts. */
  --p1-block-padding-inline: 4rem;
  --p1-block-padding-block: 1.5rem;

  /* Content widths */
  --p1-container-narrow: 48rem;
  --p1-container-default: 72rem;
  --p1-container-wide: 80rem;

  /* Radius */
  --p1-radius-sm: 0.25rem;
  --p1-radius-card: 0.75rem;
  --p1-radius-xl: 1.5rem;
  --p1-radius-pill: 9999px;

  /* Type — PDS uses ff (font-family) naming, not "family" */
  --p1-font-body: var(--pds-typography-ff-default, Inter, system-ui, sans-serif);
  --p1-font-heading: var(--pds-typography-ff-compact, var(--p1-font-body));
  --p1-text-xs: 0.75rem;
  --p1-text-sm: 0.875rem;
  --p1-text-base: 1rem;
  --p1-text-lg: 1.125rem;
  --p1-text-xl: 1.5rem;
  --p1-text-2xl: 2rem;
  --p1-text-3xl: 2.5rem;
  --p1-text-4xl: 3.5rem;
  --p1-leading-tight: 1.15;
  --p1-leading-normal: 1.6;
  --p1-leading-relaxed: 1.625;
  --p1-weight-normal: 400;
  --p1-weight-medium: 500;
  --p1-weight-semibold: 600;
  --p1-weight-bold: 700;

  /* Elevation */
  --p1-shadow-sm: 0 1px 4px oklch(0 0 0 / 0.06);
  --p1-shadow-md: 0 2px 12px oklch(0 0 0 / 0.08);
  --p1-shadow-lg: 0 8px 32px oklch(0 0 0 / 0.15);
  --p1-shadow-play: 0 4px 24px oklch(0 0 0 / 0.35);
  --p1-shadow-card-featured: 0 8px 32px oklch(0 0 0 / 0.2);
  --p1-drop-shadow-text: 0 1px 4px oklch(0 0 0 / 0.6);
}

This file installs into your project as app/p1-tokens.css. Override any --p1-* variable to retheme every block at once.
